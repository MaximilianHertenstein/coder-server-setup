===== setup-coder.sh =====
     1	#!/usr/bin/env bash
     2	# ============================================================================
     3	# Coder-Server Setup – coder.ghse-unterricht.de
     4	# Zielsystem: Debian 13 (trixie)
     5	#
     6	# Aufruf als root (ohne root bricht das Skript vor jeder Änderung ab):
     7	#   chmod +x setup-coder.sh
     8	#   sudo ./setup-coder.sh
     9	# Persistente Werte (Domain, ACME-Mail, fail2ban-Admin-IP) in setup.conf
    10	# ablegen (Vorlage: setup.conf.example, gitignored) – die überlebt Re-Runs.
    11	# Für einen Lauf gilt weiterhin: sudo env DOMAIN_CODER=… ./setup-coder.sh
    12	# (Umgebung schlägt setup.conf, setup.conf schlägt den Default).
    13	# Einzige interaktive Eingabe: Erstlauf ohne vorhandenes Konto, wenn
    14	# CODER_FIRST_USER_EMAIL und CODER_FIRST_USER_USERNAME gesetzt und stdin ein
    15	# TTY sind, fragt das Skript einmal unsichtbar das Erstkonto-Passwort ab
    16	# (read -rs; leer lassen = Konto später im Browser anlegen). Sonst kein
    17	# Prompt. Nicht-interaktiv: CODER_FIRST_USER_PASSWORD setzen (OPERATIONS §11).
    18	#
    19	# Mehrfach ausführbar (idempotent): bei Abbruch, fehlendem DNS oder fehlender
    20	# Portweiterleitung einfach erneut laufen lassen; fehlendes DNS meldet der
    21	# Abschlusscheck als WARNUNG, nicht als Fehler.
    22	#
    23	# Inhaltliche Details bewusst nur in den Dokumenten (Single Source):
    24	#   README.md        Start, Voraussetzungen (DNS/Wildcard), Prüfstand
    25	#   OPERATIONS.md    Betrieb, Wartungsskripte, Cron, Netz-Anforderung, Updates
    26	#   SECURITY.md      Socket-Risiko, D-SSH, On-Demand-TLS-Budget, Monitoring
    27	#   RESTORE.md       Wiederherstellung
    28	# Image-Versionen und Pin-Regeln stehen an den Variablendefinitionen unten.
    29	# ============================================================================

    31	set -euo pipefail
    32	
    33	# Eigenes Verzeichnis: die installierbaren Wartungsskripte liegen als echte
    34	# Dateien unter scripts/ (direkt per bash -n/shellcheck prüfbar) und werden
    35	# über write_managed von dort eingespielt. Voraussetzung: das Repo liegt
    36	# vollständig auf dem Server (git clone, siehe README).
    37	SETUP_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
    38	
    39	# Root ist Pflicht – zuerst, vor jeder Prüfung und jeder Änderung: Lock,
    40	# Rechte und Dateien unter /opt/coder gehören root, und die Wartungsskripte
    41	# sperren dieselbe Datei. Ohne root würde das Setup scheitern, nachdem es
    42	# bereits Pakete installiert hätte.
    43	# sudo leitet fremde Umgebungsvariablen nicht durch; Werte gehören hinter
    44	# das sudo: sudo env DOMAIN_CODER=… ACME_EMAIL=… ./setup-coder.sh
    45	if [ "$(id -u)" -ne 0 ]; then
    46	  echo "Fehler: dieses Setup muss als root laufen – Aufruf: sudo ./setup-coder.sh" >&2
    47	  echo "Betreiberwerte dauerhaft in setup.conf eintragen (Vorlage: setup.conf.example);" >&2
    48	  echo " für einen Lauf gehören Variablen hinter das sudo, sonst greifen die Defaults:" >&2
    49	  echo "  sudo env DOMAIN_CODER=coder.beispiel.de ACME_EMAIL=admin@beispiel.de ./setup-coder.sh" >&2
    50	  exit 1
    51	fi
    52	
    53	# Zielplattform vor jeder Serveränderung prüfen. Die Docker-APT-Quelle unten
    54	# übernimmt VERSION_CODENAME aus /etc/os-release; ohne Guard könnte ein Lauf
    55	# auf Ubuntu oder Debian 12 eine unpassende Quelle eintragen und Pakete
    56	# verändern, obwohl nur Debian 13 (trixie) unterstützt wird.
    57	if [ ! -r /etc/os-release ]; then
    58	  echo "Fehler: /etc/os-release fehlt oder ist nicht lesbar; Zielplattform nicht prüfbar. Abbruch ohne Serveränderungen." >&2
    59	  exit 1
    60	fi
    61	. /etc/os-release
    62	if [ "${ID:-}" != "debian" ] || [ "${VERSION_CODENAME:-}" != "trixie" ]; then
    63	  printf 'Fehler: nicht unterstützte Zielplattform (ID=%s, VERSION_CODENAME=%s). Erwartet wird Debian 13 (trixie). Abbruch vor Serveränderungen.\n' \
    64	    "${ID:-unbekannt}" "${VERSION_CODENAME:-unbekannt}" >&2
    65	  exit 1
    66	fi

    68	# Das Setup schreibt Compose- und Wartungsskripte und startet den Stack.
    69	# Ein paralleler Wartungslauf könnte Dateien während ihrer Ausführung
    70	# überschreiben. Vor diesen Konfigurations- und Stack-Änderungen nimmt das
    71	# Setup denselben flock wie die Wartungsskripte: er hängt am Deskriptor 8
    72	# dieses Prozesses und bleibt gesperrt, bis alle Prozesse ihn geerbt haben,
    73	# beendet sind. Alle Aufrufe hier sind kurzlebig, kein Dauerläufer erbt ihn.
    74	ENV_TMP=""
    75	MANAGED_TMP=""
    76	setup_cleanup() {
    77	  local rc=$?
    78	  for _tmp in "${ENV_TMP}" "${MANAGED_TMP}"; do
    79	    [ -n "${_tmp}" ] && rm -f -- "${_tmp}" 2>/dev/null || true
    80	  done
    81	  return "${rc}"
    82	}

    84	mkdir -p /opt/coder
    85	sh -c 'umask 077; : >> /opt/coder/.maintenance.lock; chown root:root /opt/coder/.maintenance.lock; chmod 600 /opt/coder/.maintenance.lock'
    86	exec 8>>/opt/coder/.maintenance.lock
    87	if ! flock -w 300 8; then
    88	  echo "Fehler: gemeinsamer Wartungs-Lock nach 300 s nicht erhalten; Setup abgebrochen (Exit 75)." >&2
    89	  exit 75
    90	fi
    91	export CODER_MAINTENANCE_LOCK_HELD=1
    92	trap setup_cleanup EXIT
    93	
    94	# Alle Werte lassen sich beim Aufruf per Umgebungsvariable überschreiben,
    95	# ohne diese Datei zu ändern, z.B.:
    96	#   DOMAIN_CODER=andere-domain.de ./setup-coder.sh
    97	# A5: Persistente Betreiberwerte aus setup.conf (neben dem Skript, gitignored).
    98	# Ohne sie würde ein späterer Re-Run ohne "sudo env ..." die Defaults zurück-
    99	# setzen – z.B. die Admin-IP aus fail2ban ignorip. Vorrang:
   100	#   Umgebungsvariable (aktueller Lauf) > setup.conf > eingebauter Default.
   101	# Nur die vier freigegebenen Schlüssel werden gelesen, alles andere (auch
   102	# Syntaxfehler) überschreibt nichts und bricht den Lauf nicht ab; danach
   103	# bricht die Domain-Prüfung bei ungültigem DOMAIN_CODER hart ab.
   104	if [ -f "${SETUP_DIR}/setup.conf" ]; then
   105	  while IFS='=' read -r _sk _sv || [ -n "${_sk}" ]; do
   106	    case "${_sk}" in ''|\#*) continue ;; esac
   107	    # Werte wörtlich übernehmen wäre schädlich: CRLF (Windows-Editor) landet
   108	    # als \r in Domain/Jail/Compose, umschließende Anführungszeichen in der
   109	    # fail2ban- oder Caddyfile-Konfiguration. Beides hier strippen.
   110	    _sv="${_sv%$'\r'}"
   111	    case "${_sv}" in
   112	      \"*\") _sv="${_sv#\"}"; _sv="${_sv%\"}" ;;
   113	      \'*\') _sv="${_sv#\'}"; _sv="${_sv%\'}" ;;
   114	    esac
   115	    case "${_sk}" in
   116	      DOMAIN_CODER)       [ -n "${DOMAIN_CODER+x}" ] || DOMAIN_CODER="${_sv}" ;;
   117	      ACME_EMAIL)         [ -n "${ACME_EMAIL+x}" ] || ACME_EMAIL="${_sv}" ;;
   118	      ACME_CA)            [ -n "${ACME_CA+x}" ] || ACME_CA="${_sv}" ;;
   119	      FAIL2BAN_IGNORE_IP) [ -n "${FAIL2BAN_IGNORE_IP+x}" ] || FAIL2BAN_IGNORE_IP="${_sv}" ;;
   120	    esac
   121	  done < "${SETUP_DIR}/setup.conf"
   122	fi
   123	# ":=" setzt den Default nur, wenn die Variable noch nicht (von außen oder
   124	# über setup.conf) gesetzt wurde.
   125	: "${DOMAIN_CODER:=coder.ghse-unterricht.de}"
   126	: "${ACME_EMAIL:=hert@ghse.de}"
   127	# Domain hart prüfen, bevor sie in Caddyfile/Compose wandert: Leerzeichen,
   128	# Anführungszeichen, \r oder Grossbuchstaben würden die Konfiguration oder
   129	# die TLS-Ausstellung brechen.
   130	if ! [[ "${DOMAIN_CODER}" =~ ^[a-z0-9]([a-z0-9.-]*[a-z0-9])?$ ]]; then
   131	  echo "Fehler: DOMAIN_CODER='${DOMAIN_CODER}' ist keine gültige Domain (erlaubt: klein, a-z, 0-9, '.', '-'; kein Leerzeichen/Anführung/\\r)." >&2
   132	  echo "In setup.conf oder als Umgebungsvariable korrigieren, dann Setup erneut." >&2
   133	  exit 1
   134	fi
   135	# Pin-Regeln: CODER/PROMETHEUS/GRAFANA/NODE_EXPORTER exakt, CADDY bewusst nur
   136	# minor-gepinnt (z. B. "2.11") – TLS-taugliche Patch-Updates sollen über
   137	# "docker compose pull" ankommen; postgres:18-alpine analog (Major-Tag).
   138	# WICHTIG: Das ghcr.io-Tag TRÄGT das "v"-Präfix (v2.37.3). Am 2026-09-28 und
   139	# erneut am 2026-10-03 gegen die Registry geprüft: ghcr.io/coder/coder:2.36.6
   140	# und :2.37.3 -> HTTP 404, mit "v" (v2.36.6/v2.37.3) -> 200; die Release-
   141	# Notes nennen das Tag teils ohne "v" (falsch, das bricht Schritt 3 ab).
   142	# Vor jeder Versionserhöhung IMMER per "docker pull"
   143	# gegen die Registry belegen, nicht nach dem Wortlaut der Notes. Für einen
   144	# bit-genauen, reproduzierbaren Pin bräuchte man ein Image-Digest (sha256:...).
   145	# Stand 2026-10-03 (Upstream-Stable geprüft: releases/latest bzw.
   146	# prerelease-Flags): Coder v2.37.3 (v2.36.6 war die Vorgängerlinie),
   147	# Caddy 2.11 (2.11.7 released), Prometheus v3.15.0, Grafana 12.4.12
   148	# (12.x-Security-Backports; Major 13 bewusst nicht), node-exporter
   149	# v1.12.1 = aktuell, Postgres 19 nur Beta -> 18-alpine bleibt.
   150	: "${CODER_VERSION:=v2.37.3}"              # https://github.com/coder/coder/releases prüfen
   151	: "${CADDY_VERSION:=2.11}"                # https://hub.docker.com/_/caddy/tags
   152	: "${PROMETHEUS_VERSION:=v3.15.0}"        # https://github.com/prometheus/prometheus/releases
   153	: "${GRAFANA_VERSION:=12.4.12}"           # https://github.com/grafana/grafana/releases
   154	: "${NODE_EXPORTER_VERSION:=v1.12.1}"     # https://github.com/prometheus/node_exporter/releases
   155	# Admin-Adressen, die fail2ban nie sperren darf – NUR echte Admin-Plätze hier
   156	# eintragen, ausdrücklich NICHT die geteilte Schul-NAT: Rateversuche aus
   157	# dem Schulnetz sollen nach 5 Fehlversuchen sperren. Bewusst akzeptiert:
   158	# die gemeinsame NAT-IP kann dann gesperrt sein (Recovery: anderes Netz
   159	# oder Proxmox-Konsole). Default bleibt loopback.
   160	: "${FAIL2BAN_IGNORE_IP:=127.0.0.1/8 ::1}"

   162	# ---- Wirksam gewesene Version + Downgrade-Schutz (A5) ---------------------
   163	# Gepinnt und geschützt wird nur CODER_VERSION: allein Coder führt
   164	# DB-Schema-Migrationen aus, ein stilles Zurücksetzen wäre dort irreversibel.
   165	# Caddy/Prometheus/node-exporter sind beim Restarten zustandslos, ihr
   166	# Versionswechsel ist ein bewusster Compose-Tag-Eintrag; Grafana-Downgrades
   167	# sind wegen der lokalen SQLite-Daten bewusste Manuellearbeit (Release-Notes
   168	# lesen). POSTGRES_VERSION wird gar nicht verwaltet: Postgres-Major nur
   169	# manuell ändern, immer mit frischem Backup.
   170	# Versions.env: nur Version, keine Geheimnisse -> 644, ohne sudo lesbar.
   171	# Downgrade-Versuch hart abgebrochen, sofern nicht ausdrücklich
   172	# CODER_ALLOW_DOWNGRADE=yes gesetzt ist (bewusster Rollback, siehe
   173	# RESTORE.md). Eine Änderung NACH oben (Upgrade) bleibt ohne Rückfrage.
   174	VERSIONS_FILE="/opt/coder/versions.env"
   175	VERSION_VARS="CODER_VERSION"
   176	VERSIONS_PREV="$(cat "${VERSIONS_FILE}" 2>/dev/null || true)"
   177	for _vv in ${VERSION_VARS}; do
   178	  _vnew="${!_vv:-}"
   179	  _vprev="$(printf '%s\n' "${VERSIONS_PREV}" | awk -F= -v k="${_vv}" '$1 == k {print substr($0, index($0, "=") + 1); exit}')"
   180	  _vprev_sort="${_vprev#v}"
   181	  _vnew_sort="${_vnew#v}"
   182	  if [ -n "${_vprev_sort}" ] && [ -n "${_vnew_sort}" ] && [ "${_vprev_sort}" != "${_vnew_sort}" ] \
   183	    && [ "$(printf '%s\n%s\n' "${_vprev_sort}" "${_vnew_sort}" | sort -V | head -n 1)" = "${_vnew_sort}" ]; then
   184	    echo "Fehler: Downgrade erkannt – ${_vv} würde von ${_vprev} auf ${_vnew} zurückgesetzt." >&2
   185	    echo "Ein Downgrade kann bei Coder eine bereits ausgeführte DB-Schema-Migration" >&2
   186	    echo "nicht rückgängig machen und die Installation unbrauchbar machen." >&2
   187	    if [ "${CODER_ALLOW_DOWNGRADE:-}" = "yes" ]; then
   188	      echo "CODER_ALLOW_DOWNGRADE=yes gesetzt – bewusstes Downgrade wird durchgeführt." >&2
   189	    else
   190	      echo "Abbruch, nichts wurde verändert. Bewusstes Downgrade (mit verifiziertem" >&2
   191	      echo "Dump, siehe RESTORE.md):" >&2
   192	      echo "  sudo env CODER_ALLOW_DOWNGRADE=yes CODER_VERSION=... ./setup-coder.sh" >&2
   193	      exit 1
   194	    fi
   195	  fi
   196	done

   202	declare -a SUMMARY=()
   203	ERROR_COUNT=0
   204	ok()   { echo "  [OK]     $1"; SUMMARY+=("[OK]     $1"); }
   205	warn() { echo "  [WARNUNG] $1"; SUMMARY+=("[WARNUNG] $1"); }
   206	fail() { echo "  [FEHLER] $1"; SUMMARY+=("[FEHLER] $1"); ERROR_COUNT=$((ERROR_COUNT + 1)); }

   208	check_container_running() {
   209	  local name="$1"
   210	  local label="$2"
   211	  if [ "$(docker inspect -f '{{.State.Running}}' "${name}" 2>/dev/null || echo false)" = "true" ]; then
   212	    ok "${label} läuft"
   213	  else
   214	    fail "${label} läuft NICHT – prüfen mit: sudo docker logs ${name}"
   215	  fi

   218	# Das Setup ist deklarativer Besitzer der generierten Dateien, das
   219	# Überschreiben von Handänderungen ist in README/OPERATIONS dokumentiert.
   220	# Was zählt: Temp schreiben, validieren, atomar ersetzen.
   221	
   222	# Atomar eine vom Setup verwaltete Datei schreiben. Die temporäre Datei liegt
   223	# im Zielverzeichnis, damit mv dieselbe Dateisystem-Grenze nicht überschreitet.
   224	# Bei identischem Inhalt bleibt der Ziel-Inode unangetastet; das vermeidet
   225	# unnötige Reloads und schützt laufende Leser. Modus wird nachgezogen.
   226	# MANAGED_TMP wird vom EXIT-Cleanup entfernt, falls der Lauf vor
   227	# dem mv abbricht.
   228	# A2: Optionaler Validierer ($4, Funktionsname) läuft auf der TEMP-Datei, VOR
   229	# dem Austausch der Live-Datei. Bei Fehlschlag bleibt die vorhandene Datei
   230	# unangetastet und der Lauf bricht ab – eine ungültige Konfiguration landet
   231	# damit nie auf der Platte und könnte den nächsten Reboot nicht überleben.
   232	write_managed() { # $1=Pfad $2=Modus $3=Referenz-Label (Reserve, ungenutzt) [$4=Validierer]; Inhalt via stdin
   233	  local file="$1" mode="$2" validator="${4:-}" dir base
   234	  dir="${file%/*}"
   235	  base="${file##*/}"
   236	  MANAGED_TMP=$(mktemp "${dir}/.${base}.tmp.XXXXXX")
   237	  if ! cat > "${MANAGED_TMP}"; then
   238	    rm -f -- "${MANAGED_TMP}" 2>/dev/null || true
   239	    MANAGED_TMP=""
   240	    return 1
   241	  fi
   242	  case "${file}" in
   243	    *.sh)
   244	      if ! bash -n "${MANAGED_TMP}"; then
   245	        echo "Fehler: generiertes Skript ist syntaktisch ungültig: ${file}; vorhandene Datei bleibt erhalten." >&2
   246	        rm -f -- "${MANAGED_TMP}"
   247	        MANAGED_TMP=""
   248	        return 1
   249	      fi
   250	      ;;
   251	  esac
   252	  if [ -n "${validator}" ]; then
   253	    # mktemp legt 600 an; promtool laeuft im Container als nobody und darf die
   254	    # Temp-Datei dann nicht lesen (live gefunden 2026-10-01). Vor dem
   255	    # Validieren auf den spaeteren Endmodus heben, danach chmod "${mode}".
   256	    chmod 644 -- "${MANAGED_TMP}"
   257	    if ! "${validator}" "${MANAGED_TMP}"; then
   258	      echo "Fehler: Validierung fehlgeschlagen (${validator}): ${file} – Live-Datei bleibt unverändert, Lauf abgebrochen." >&2
   259	      rm -f -- "${MANAGED_TMP}" 2>/dev/null || true
   260	      MANAGED_TMP=""
   261	      return 1
   262	    fi
   263	  fi
   264	  chmod "${mode}" "${MANAGED_TMP}"
   265	  if cmp -s -- "${MANAGED_TMP}" "${file}"; then
   266	    chmod "${mode}" "${file}"
   267	    rm -f -- "${MANAGED_TMP}"
   268	  else
   269	    mv -f -- "${MANAGED_TMP}" "${file}"
   270	  fi
   271	  MANAGED_TMP=""
   272	}

   274	# B-013: Retry max. 3 mit Schlaf + Ursachen-Log (Schulnetz!). Keine endlosen
   275	# Retries, keine Retry-Logik um LE-Ausstellung (Budget) – nur apt/curl/pull.
   276	# Gibt den finalen Exit-Code zurück (set -e bricht danach wie bisher ab).
   277	retry3() {
   278	  local attempt=1
   279	  while true; do
   280	    if "$@"; then
   281	      return 0
   282	    elif [ "${attempt}" -ge 3 ]; then
   283	      echo "FEHLER nach 3 Versuchen: $* – Ursache prüfen (Netz/DNS/Registry), dann Setup erneut." >&2
   284	      return 1
   285	    else
   286	      echo "Versuch ${attempt} fehlgeschlagen ($*), warte 10 s vor Wiederholung ..." >&2
   287	      sleep 10
   288	      attempt=$((attempt + 1))
   289	    fi
   290	  done
   291	}

   293	# A2: Validierer für write_managed – prüfen die TEMP-Datei (Pfad $1), bevor die
   294	# Live-Datei ausgetauscht wird. Die caddy-/promtool-Präfixe brauchen die
   295	# Images, deshalb wird vor dem Schreiben dieser Dateien gepullt (siehe unten).
   296	# caddy:2.11 hat KEINEN ENTRYPOINT (nur CMD ["caddy","run",...]) – ohne
   297	# --entrypoint würde "validate" selbst als Binary ausgeführt und scheitern
   298	# mit 'executable file not found in $PATH'.
   299	validate_compose() {
   300	  docker compose -f "$1" config --quiet
   301	}
   302	validate_caddyfile() {
   303	  docker run --rm --entrypoint caddy -v /opt/coder/caddy:/etc/caddy:ro "caddy:${CADDY_VERSION}" \
   304	    validate --config "/etc/caddy/$(basename "$1")" --adapter caddyfile
   305	}
   306	validate_prom_rules() {
   307	  docker run --rm --entrypoint /bin/promtool -v /opt/coder/prometheus:/etc/prometheus:ro "prom/prometheus:${PROMETHEUS_VERSION}" \
   308	    check rules "/etc/prometheus/rules/$(basename "$1")"
   309	}
   310	validate_prom_config() {
   311	  docker run --rm --entrypoint /bin/promtool -v /opt/coder/prometheus:/etc/prometheus:ro "prom/prometheus:${PROMETHEUS_VERSION}" \
   312	    check config "/etc/prometheus/$(basename "$1")"
   313	}

   315	# A10: Alle SSH-Vorbedingungen HIER, vor Docker-Installation und Stack-Start.
   316	# Die Firewall-Regel unten setzt die hier erkannten sshd-Ports; ohne lesbare
   317	# sshd-Konfiguration wäre unklar, welchen Port "ufw enable" noch offen lässt –
   318	# deshalb hart abbrechen, bevor irgendetwas installiert oder verändert wird
   319	# (Zugriff dann nur noch über Konsolenzugang des Hosts).
   320	if ! command -v sshd > /dev/null 2>&1; then
   321	  echo "Fehler: sshd nicht gefunden – openssh-server fehlt; ohne SSH-Zugang würde die Firewall-Änderung den Host absperren." >&2
   322	  echo "openssh-server installieren (apt install openssh-server), dann Setup erneut." >&2
   323	  exit 1
   324	fi
   325	if ! SSHD_EFFECTIVE_CONFIG="$(sshd -T 2>/dev/null)"; then
   326	  echo "Fehler: sshd-Konfiguration konnte nicht gelesen werden – Abbruch vor jeder Serveränderung." >&2
   327	  exit 1
   328	fi
   329	SSHD_PORTS="$(printf '%s\n' "${SSHD_EFFECTIVE_CONFIG}" | awk '/^port /{print $2}' | sort -u | xargs echo 2>/dev/null || true)"
   330	if [ -z "${SSHD_PORTS}" ]; then
   331	  echo "Fehler: sshd -T lieferte keinen Port – Abbruch vor jeder Serveränderung." >&2
   332	  exit 1
   333	fi

   338	# C1: Alle Cron-Zeiten dieses Skripts (Backup 01:00, Reboot 03:00)
   339	# gelten in LOCALTIME. Neue VMs laufen oft auf UTC, dann läuft das
   340	# Backup um 03:00 statt 01:00 Berlin und der Reboot mitten in der Nacht.

   348	      echo "Hinweis: Zeitzone konnte nicht gesetzt werden – Cron-Zeiten gelten in '${_tz:-unbekannt}' (C1)." >&2
   349	    fi

   731	# Caddyfile – echtes, öffentlich vertrauenswürdiges Let's-Encrypt-Zertifikat,
   732	# inklusive Wildcard-Subdomain für Coder-Workspace-Apps (Port-Forwarding,
   733	# code-server-Vorschau etc. laufen über eigene Subdomains).
   734	#
   735	# Wildcard-Zertifikate lassen sich nicht per normaler HTTP-01-Challenge
   736	# ausstellen (die validiert immer nur EINEN konkreten Hostnamen). Caddys
   737	# On-Demand-TLS holt das Zertifikat beim ersten Aufruf einer neuen Subdomain
   738	# (5-30 s länger beim allerersten Aufruf). "Echte" Wildcard-Zertifikate ohne
   739	# diese Verzögerung bräuchten einen eigenen Caddy-Build mit DNS-Provider-
   740	# Plugin (z.B. Cloudflare/Route53) – dafür fehlt ein API-Token (D-TLS).
   741	#
   742	# Aufbau (B1):
   743	#   - Global: On-Demand-TLS mit ask-Endpunkt auf 127.0.0.1:9123.
   744	#   - :9123: der ask-Endpunkt selbst. REIN innerhalb des Containers (der Port
   745	#     steht nicht in "ports:" der compose-Datei), also ohne ufw-Regel und von
   746	#     außen nicht erreichbar. Antwort 200 = Zertifikat erlaubt, 404 = verboten.
   747	#   - Apex (${DOMAIN_CODER}): OHNE on_demand – Zertifikat beim Start über
   748	#     HTTP-01, Erneuerung durch Caddy. Berührt das On-Demand-Budget nicht.
   749	#   - Wildcard (*.${DOMAIN_CODER}): MIT on_demand, der ask-Endpunkt prüft das
   750	#     Muster. Ein Angreifer bekommt damit kein Zertifikat für beliebige
   751	#     Namen; für Namen IM Muster kann er es weiterhin auslösen – das Muster
   752	#     ist ein grober Filter, kein Budget-Schutz (P-005, Dokumentation in
   753	#     SECURITY.md). Wochenbudget 50/Woche pro registrierter Domain bleibt bis
   754	#     D-TLS (DNS-01-Wildcard ohne On-Demand) das Thema.
   755	#     Eine automatische Budget-Überwachung gibt es nicht (manuelle
   756	#     Sichtprüfung) – Erschöpfung fällt als HTTPS-Fehler auf;
   757	#     Gegenmittel ist DNS-01, sobald Zone/Token geklärt sind (D-TLS).
   758	#
   759	# Muster (Quelle: Coder-Quellcode coderd/workspaceapps/appurl/appurl.go,
   760	# Format "{PORT/APP_SLUG}[--{AGENT_NAME}]--{WORKSPACE_NAME}--{USERNAME}":
   761	#     8080--main--myworkspace--john.${DOMAIN_CODER}         (App/Port-Forward)
   762	#     33295s--agent--workspace--user--apps.${DOMAIN_CODER}  (HTTPS-Port)
   763	# Erste Label muss also mindestens zwei "--"-Trenner tragen und nur aus
   764	# Buchstaben/Ziffern/Einfach-Bindestrich bestehen. Punkte in der ersten Label
   765	# sind ausgeschlossen – damit bleiben "evil.${DOMAIN_CODER}",
   766	# "a.b.${DOMAIN_CODER}" und der Apex selbst draußen. Richtige Warnung bei
   767	# unbekannter Coder-Version: Muster gegen die Coder-Doku prüfen, nicht den
   768	# Code raten lassen.
   769	# DOMAIN_CODER_REGEX enthaelt KEINE Backslashes in den Punktgruppen ([.]
   770	# statt "\.") und KEIN "$"-Escaping-Probleme, weil CEL-Strings in Caddy dort
   771	# keine Escape-Sequenzen kennen – deshalb die Klammern-Variante.
   772	DOMAIN_CODER_ESC="$(printf '%s' "${DOMAIN_CODER}" | sed 's/\./[.]/g')"
   773	DOMAIN_CODER_REGEX="^[a-zA-Z0-9]+(-[a-zA-Z0-9]+)*(--[a-zA-Z0-9]+(-[a-zA-Z0-9]+)*){2,}[.]${DOMAIN_CODER_ESC}\$"
   774	# C9: Optionaler eigener ACME-Endpunkt (ACME_CA), z.B. um gegen das
   775	# Let's-Encrypt-Staging zu testen, ohne das Wochenbudget der Produktions-CA zu
   776	# verbrauchen. Leer = Caddys Standard. Die Zeile entsteht nur, wenn gesetzt –
   777	# ein leeres "acme_ca" ohne Argument wäre ungültig. Gehört in den GLOBALen
   778	# Block, nicht in "issuer acme { }": dort ist "acme_ca" kein gültiges
   779	# Property (mit caddy:2.11 nachgeprüft: "unrecognized ACME issuer property"),
   780	# während die globale Option in beide Issuer (Apex + On-Demand) durchschlägt.
   781	CADDY_ACME_CA=""
   782	if [ -n "${ACME_CA:-}" ]; then
   783	  CADDY_ACME_CA="    acme_ca ${ACME_CA}"
   784	fi

===== scripts/backup.sh =====
     1	#!/bin/bash
     2	# Tägliches Postgres-Backup (F-005/F-008/F-009, mit P-013/P-017-Korrekturen).
     3	# Aufruf: /opt/coder/backup.sh (direkt, per Cron, aus dem Setup-Pre-Upgrade-
     4	# bzw. Test-Lauf oder aus reset-coder.sh).
     5	# Gleichzeitige Läufe werden per flock serialisiert: der zweite Lauf WARTET
     6	# bis zu 300 s und fertigt danach ein eigenes Backup an, statt die Datei des
     7	# ersten per ">" zu trunkieren (F-009). Läuft der Vorgänger länger, gibt es
     8	# Exit 75 (EX_TEMPFAIL) statt des alten Exit 0.
     9	# A3: Exit 0 war der Fehler – "übersprungen" wurde als Erfolg gewertet und
    10	# ein Update/Reset lief trotz fehlendem Backup weiter. Heute liefern alle
    11	# Ausstiege (Lock-Timeout, Wartungs-Lock, Postgres nicht bereit) ≠0; die
    12	# Aufrufer (Setup, reset-coder.sh) brechen bei ≠0 mit set -e bzw. eigener
    13	# Prüfung korrekt ab.
    14	# Der gemeinsame Lock unten verhindert, dass ein eigenständiges Backup mit
    15	# Update, Passwortwechsel oder Reset überlappt. pg_dump selbst liefert zudem
    16	# einen konsistenten Snapshot.
    17	set -euo pipefail
    18	umask 077
    19	# Cron-PATH ist minimal (/usr/bin:/bin): explizit setzen (P-017), damit
    20	# docker/gzip/date/flock/logger/df/stat gefunden werden.
    21	export PATH="/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin"
    22	BACKUP_DIR="/opt/coder/backups"

   180	# sie sind kein Hindernis für die spätere volume rm. Geprüft werden nur
   181	# Nicht-Projekt-Container (fremde Workspaces), die das Volume auch nach 'down'
   182	# weiterhin halten würden. Ohne diese Einschränkung würde jeder Re-Run auf
   183	# laufendem Stack abbrechen, weil coder-postgres postgres_data hält (gemessen:
   184	# "docker ps -a --filter volume=<name>" findet laufende Container). "-q"
   185	# fehlt absichtlich neben "--format": docker ignoriert --format bei -q
   186	# ("Ignoring custom format, because both --format and --quiet are set"), das
   187	# Label käme nie an und auch Projekt-Container würden als Beleg zählen.
   188	while IFS= read -r VOL_NAME; do
   189	  [ -n "${VOL_NAME}" ] || continue
   190	  if USERS=$(docker ps -a --filter "volume=${VOL_NAME}" \
   191	      --format '{{.ID}} {{.Label "com.docker.compose.project"}}' 2>/dev/null \
   192	      | awk -v p="${PROJECT}" '$2 != p {print $1}') && [ -n "${USERS:-}" ]; then
   193	    echo "Fehler: Volume ${VOL_NAME} wird noch von Container(n) benutzt: $(echo "${USERS}" | tr '\n' ' ')" >&2
   194	    echo "Abbruch VOR 'docker compose down' – nichts angefasst: alle Container laufen weiter, Volumes und .env sind unverändert (fremde Container erst aufräumen, dann Reset erneut)." >&2
   195	    exit 1
   196	  fi
   197	done <<< "${VOLUMES_TO_DELETE}"
   198	
   199	echo "Stoppe Container (caddy_data/caddy_config bleiben unangetastet)..."
   200	echo "(Das Inventar oben war leer – Nicht-Compose-/Workspace-Container sind"
   201	echo "per harter Regel vorher entfernt worden.)"
   202	docker compose down --remove-orphans
   203	
   204	while IFS= read -r VOL_NAME; do
   205	  [ -n "${VOL_NAME}" ] || continue
   206	  docker volume rm "${VOL_NAME}"
   207	done <<< "${VOLUMES_TO_DELETE}"
   208	
   209	echo "Entferne generierte Passwörter (.env) und versions.env..."
   210	# versions.env mit: die Datei hält den wirksam gewesenen Versions-Stand, der
   211	# den Downgrade-Guard im Setup steuert – nach einem Reset ist die DB frisch,
   212	# ein restlicher Stand würde einen Neuaufbau mit niedrigerer Version blockieren.
   213	rm -f .env versions.env
   214	
   215	echo "------------------------------------------------------------"
   216	echo "Reset abgeschlossen. Zum Neuaufbau (im Repo-Verzeichnis – das Setup"
   217	echo "liegt nicht in /opt/coder):"
   218	echo "  sudo ./setup-coder.sh"
   219	echo "(erzeugt neue Passwörter und versions.env, startet den Stack komplett neu)"
   220	echo "(Workspace-/fremde Container gab es vorher nicht – harte Regel oben.)"
   221	echo "------------------------------------------------------------"
